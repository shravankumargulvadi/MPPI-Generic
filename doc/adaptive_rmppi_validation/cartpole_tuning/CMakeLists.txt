cmake_minimum_required(VERSION 3.24)
project(AdaptiveCartpoleStudy LANGUAGES C CXX CUDA)
enable_testing()
add_subdirectory(/usr/src/googletest googletest)
add_subdirectory(/repo mppi)
foreach(HORIZON IN ITEMS 100 150)

  target_link_libraries(adaptive_cartpole_h${HORIZON} PRIVATE MPPI::MPPI)
  target_compile_definitions(adaptive_cartpole_h${HORIZON} PRIVATE ADAPTIVE_CARTPOLE_HORIZON=${HORIZON})
endforeach()
